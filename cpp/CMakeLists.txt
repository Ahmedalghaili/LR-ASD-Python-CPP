cmake_minimum_required(VERSION 3.18)
project(lr_asd_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Torch REQUIRED)
find_package(OpenCV REQUIRED)
add_executable(lr_asd src/main.cpp)
target_link_libraries(lr_asd PRIVATE ${TORCH_LIBRARIES} ${OpenCV_LIBS})
target_compile_options(lr_asd PRIVATE -O3)
set_property(TARGET lr_asd PROPERTY CXX_STANDARD 17)
